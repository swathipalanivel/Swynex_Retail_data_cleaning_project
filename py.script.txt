
import pandas as pd

# 1. Load the raw dataset
df = pd.read_excel("Online Retail.xlsx")

# 2. Check the original data
print("Original rows:", len(df))
print("\nMissing values:")
print(df.isnull().sum())

print("\nDuplicate rows:", df.duplicated().sum())
print("\nData types:")
print(df.dtypes)

# 3. Remove exact duplicate records
df = df.drop_duplicates().copy()

# 4. Standardize text columns
for col in ["Description", "Country", "InvoiceNo", "StockCode"]:
    df[col] = df[col].astype("string").str.strip()

# 5. Correct data types
df["InvoiceDate"] = pd.to_datetime(
    df["InvoiceDate"], errors="coerce"
)

df["Quantity"] = pd.to_numeric(
    df["Quantity"], errors="coerce"
)

df["UnitPrice"] = pd.to_numeric(
    df["UnitPrice"], errors="coerce"
)

# 6. Remove rows with invalid essential fields
df = df.dropna(
    subset=["InvoiceNo", "StockCode",
            "InvoiceDate", "Quantity", "UnitPrice"]
).copy()

# 7. Create a revenue column
df["Revenue"] = df["Quantity"] * df["UnitPrice"]

# 8. Save the cleaned dataset
df.to_csv("cleaned_retail_data.csv", index=False)

# 9. Validate the result
print("\nCleaned rows:", len(df))
print("Remaining duplicates:", df.duplicated().sum())
print("\nRemaining missing values:")
print(df.isnull().sum())
print("\nFinal data types:")
print(df.dtypes)
